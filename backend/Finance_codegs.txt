/**
 * JSON / JSONP válaszgenerálás (Booklist minta alapján).
 */
function createJsonOrJsonpResponse_(obj, callback) {
  const json = JSON.stringify(obj);
  if (callback) {
    return ContentService
      .createTextOutput(callback + "(" + json + ");")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService
    .createTextOutput(json)
    .setMimeType(ContentService.MimeType.JSON);
}
function generatePrefixedId_(prefix) {
  // 10 hosszú alfanumerikus (0-9, A-F) – UUID-ból vágva
  const code = Utilities.getUuid().replace(/-/g, "").toUpperCase().slice(0, 10);
  return `${prefix}-${code}`;
}

function login_(email, password) {
  const user = getUserByEmail_(email);
  if (!user) return { success: false, error: "Hibás email vagy jelszó." };

  // app_access ellenőrzés (megtartva a jelenlegi logikát)
  const access = String(user.app_access || "").trim().toLowerCase();
  const hasAccess = (access === "x" || access === "1" || access === "true" || access === "yes");
  if (!hasAccess) return { success: false, error: "Nincs jogosultság (app_access)." };

  const stored = String(user.password || "").trim();
  const incoming = String(password || "");

  // Első belépés: még nincs jelszó beállítva
  if (!stored) {
    if (!incoming) {
      return { success: false, setup_required: true, message: "Első belépés: állíts be jelszót." };
    }
    // jelszó beállítása hash-elve
    setUserPasswordHash_(user, incoming);

    const token = createSessionToken_(user);
    return { success: true, token, user: { id: user.id, name: user.name, email: user.email } };
  }

  // Normál belépés (hash vagy legacy)
  const ok = verifyPassword_(stored, incoming);
  if (!ok) return { success: false, error: "Hibás email vagy jelszó." };

  // Legacy migráció: ha plain-text volt, sikeres belépés után hash-eljük
  if (!isHashedPassword_(stored)) {
    setUserPasswordHash_(user, incoming);
  }

  const token = createSessionToken_(user);
  return { success: true, token, user: { id: user.id, name: user.name, email: user.email } };
}



function logout_(token) {
  if (!token) return { success: true };
  CacheService.getScriptCache().remove("sess:" + token);
  return { success: true };
}
function requireAuth_(p) {
  const token = String(p.token || "").trim();
  if (!token) throw new Error("Nincs bejelentkezés (token hiányzik).");

  const raw = CacheService.getScriptCache().get("sess:" + token);
  if (!raw) throw new Error("A munkamenet lejárt vagy érvénytelen. Jelentkezz be újra.");

  const sess = JSON.parse(raw);
  return sess; // {id,name,email}
}
function requireAdmin_(auth) {
  const u = getUserByEmail_(auth.email);
  if (!u) throw new Error("Admin ellenőrzés: felhasználó nem található.");

  // Kötelező oszlop: is_admin
  if (typeof u.is_admin === "undefined") {
    throw new Error('Hiányzik az "is_admin" oszlop a Users lapon.');
  }

  const v = String(u.is_admin || "").trim().toLowerCase();
  const ok = (v === "x" || v === "1" || v === "true" || v === "yes");
  if (!ok) throw new Error("Nincs admin jogosultság.");
}
function getUserAccess_(email, function_key) {
  const sheet = SpreadsheetApp.getActive().getSheetByName("Permissions");
  if (!sheet) return "none";

  const em = String(email || "").trim().toLowerCase();
  const fk = String(function_key || "").trim().toLowerCase();
  if (!em || !fk) return "none";

  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return "none";

  const header = (data[0] || []).map(h => String(h || "").trim());
  const col = (name) => header.indexOf(name);

  const cEmail = col("email");
  const cKey = col("function_key");
  const cAcc = col("access");
  if (cEmail === -1 || cKey === -1 || cAcc === -1) return "none";

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const rowEmail = String(row[cEmail] || "").trim().toLowerCase();
    const rowKey = String(row[cKey] || "").trim().toLowerCase();

    if (rowEmail === em && rowKey === fk) {
      const acc = String(row[cAcc] || "").trim().toLowerCase();
      return ["none", "read", "write"].includes(acc) ? acc : "none";
    }
  }
  return "none";
}

function requirePermission_(auth, function_key, level) {
  // level: "read" | "write"
  const need = String(level || "").trim().toLowerCase();
  const fk = String(function_key || "").trim();
  if (!fk) throw new Error("Hiányzó function_key.");

  // Admin mindent vihet (megtartjuk a jelenlegi admin logikát)
  try { requireAdmin_(auth); return; } catch (_) { }

  const email = (auth && auth.email) ? String(auth.email).trim().toLowerCase() : "";
  const acc = getUserAccess_(email, fk);

  const rank = { none: 0, read: 1, write: 2 };
  if ((rank[acc] || 0) < (rank[need] || 0)) {
    throw new Error("Nincs jogosultság (" + fk + ":" + need + ").");
  }
}


function createSessionToken_(user) {
  const token = Utilities.getUuid();
  const sess = { id: user.id, name: user.name, email: user.email };

  // 12 óra (másodpercben)
  CacheService.getScriptCache().put("sess:" + token, JSON.stringify(sess), 60 * 60 * 12);
  return token;
}
function isHashedPassword_(stored) {
  return /^sha256\$/i.test(String(stored || "").trim());
}

function makeSalt_() {
  return Utilities.getUuid().replace(/-/g, "");
}

function toHex_(bytes) {
  return bytes.map(b => ("0" + (b & 0xff).toString(16)).slice(-2)).join("");
}

// Formátum: sha256$<salt>$<hexhash>
function hashPassword_(plain, salt) {
  const pw = String(plain ?? "");
  const s = String(salt ?? "");
  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    s + pw,
    Utilities.Charset.UTF_8
  );
  return `sha256$${s}$${toHex_(bytes)}`;
}

function verifyPassword_(stored, incomingPlain) {
  const st = String(stored || "").trim();
  const pw = String(incomingPlain ?? "");

  if (!isHashedPassword_(st)) return st === pw; // legacy/plain-text

  const parts = st.split("$");
  if (parts.length !== 3) return false;

  const salt = parts[1];
  const hex = parts[2];
  const recomputed = hashPassword_(pw, salt);
  return recomputed === `sha256$${salt}$${hex}`;
}

function setUserPasswordHash_(user, newPlainPassword) {

  const salt = makeSalt_();
  const hashed = hashPassword_(newPlainPassword, salt);

}

function getUserByEmail_(email) {
  const em = String(email || "").trim().toLowerCase();
  if (!em) return null;

  const sheet = SpreadsheetApp.getActive().getSheetByName("Users");
  if (!sheet) throw new Error('Nem található a "Users" munkalap.');

  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return null;

  const header = data[0].map(h => String(h || "").trim());
  const col = (name) => header.indexOf(name);

  const cId = col("id");
  const cName = col("name");
  const cEmail = col("email");
  const cPw = col("password");
  const cAcc = col("app_access");
  const cAdmin = col("is_admin");


  if (cEmail === -1 || cPw === -1 || cAcc === -1) {
    throw new Error('A "Users" fejléc hibás. Elvárt oszlopok: id, name, email, password, app_access, created_at');
  }

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const rowEmail = String(row[cEmail] || "").trim().toLowerCase();
    if (rowEmail === em) {
      return {
        id: cId !== -1 ? row[cId] : "",
        name: cName !== -1 ? row[cName] : "",
        email: row[cEmail],
        password: row[cPw],
        app_access: row[cAcc],
        is_admin: cAdmin !== -1 ? row[cAdmin] : undefined,
        _rowIndex1Based: i + 1,
        _colPw1Based: cPw + 1
      };

    }
  }
  return null;
}

function addUser_(name, email) {
  const nm = String(name || "").trim();
  const em = String(email || "").trim().toLowerCase();

  if (!nm) throw new Error("A név kötelező.");
  if (!em) throw new Error("Az e-mail kötelező.");

  const sheet = SpreadsheetApp.getActive().getSheetByName("Users");
  if (!sheet) throw new Error('Nem található a "Users" munkalap.');

  const data = sheet.getDataRange().getValues();
  const header = data[0].map(h => String(h || "").trim());
  const col = (name) => header.indexOf(name);

  const cId = col("id");
  const cName = col("name");
  const cEmail = col("email");
  const cPw = col("password");
  const cAcc = col("app_access");
  const cCreated = col("created_at");
  const cAdmin = col("is_admin"); // opcionális, de admin UI-hoz ajánlott

  if (cId === -1 || cName === -1 || cEmail === -1 || cPw === -1 || cAcc === -1 || cCreated === -1) {
    throw new Error('A "Users" fejléc hibás. Elvárt oszlopok: id, name, email, password, app_access, created_at (és opcionálisan is_admin).');
  }

  // duplikáció check
  for (let i = 1; i < data.length; i++) {
    const rowEmail = String(data[i][cEmail] || "").trim().toLowerCase();
    if (rowEmail === em) throw new Error("Már létezik ilyen e-maillel felhasználó.");
  }

  const id = generatePrefixedId_("US");
  const created_at = new Date();

  const newRow = new Array(header.length).fill("");
  newRow[cId] = id;
  newRow[cName] = nm;
  newRow[cEmail] = em;
  newRow[cPw] = "";          // üres jelszó -> első belépéskor beállítás (a jelenlegi login_ logikád szerint)
  newRow[cAcc] = "x";        // kapjon app hozzáférést
  newRow[cCreated] = created_at;
  if (cAdmin !== -1) newRow[cAdmin] = ""; // nem admin alapból

  // --- "Tranzakció-szerű" végrehajtás: user + default permissions, rollback hibára ---
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);

  // előre eltesszük, hova fogunk beszúrni
  const userRowIndex1Based = sheet.getLastRow() + 1;

  let permSheet = null;
  let permStartRow1Based = null;
  let permInsertedCount = 0;

  try {
    // 1) User beszúrás
    sheet.appendRow(newRow);

    // 2) Default jogosultságok létrehozása (minden function_key -> none)
    const fnSheet = SpreadsheetApp.getActive().getSheetByName("Functions");
    permSheet = SpreadsheetApp.getActive().getSheetByName("Permissions");

    if (!fnSheet) throw new Error('Nem található a "Functions" munkalap.');
    if (!permSheet) throw new Error('Nem található a "Permissions" munkalap.');

    const fnData = fnSheet.getDataRange().getValues();
    // ha csak fejléc van, akkor nincs mit beszúrni
    if (fnData.length > 1) {
      permStartRow1Based = permSheet.getLastRow() + 1;

      for (let i = 1; i < fnData.length; i++) {
        const functionKey = String(fnData[i][0] || "").trim(); // function_key az első oszlopban
        if (!functionKey) continue;

        const sp = setPermission_(em, functionKey, "none");
        // csak akkor számoljuk "beszúrtnak", ha nem update volt
        if (!sp || !sp.updated) permInsertedCount++;


      }
    }

    return { success: true, id };

  } catch (err) {
    // --- rollback: permissions (ha már beszúrtunk), majd user sor ---
    try {
      if (permSheet && permStartRow1Based && permInsertedCount > 0) {
        permSheet.deleteRows(permStartRow1Based, permInsertedCount);
      }
    } catch (e) { }

    try {
      // csak akkor töröljük, ha tényleg beszúrtuk: a lastRow legalább akkora, mint az elmentett index
      if (sheet.getLastRow() >= userRowIndex1Based) {
        sheet.deleteRow(userRowIndex1Based);
      }
    } catch (e) { }

    throw err;

  } finally {
    try { lock.releaseLock(); } catch (e) { }
  }

}
function getFunctions_() {
  const sheet = SpreadsheetApp.getActive().getSheetByName("Functions");
  if (!sheet) throw new Error('Nem található a "Functions" munkalap.');

  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return { success: true, functions: [] };

  const header = data[0].map(h => String(h || "").trim());
  const col = (name) => header.indexOf(name);

  const cKey = col("function_key");
  const cName = col("name");
  const cDesc = col("description");

  if (cKey === -1 || cName === -1 || cDesc === -1) {
    throw new Error('A "Functions" fejléc hibás. Elvárt oszlopok: function_key, name, description');
  }

  const functions = [];
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const key = String(row[cKey] || "").trim();
    if (!key) continue;

    functions.push({
      function_key: key,
      name: String(row[cName] || "").trim(),
      description: String(row[cDesc] || "").trim()
    });
  }

  return { success: true, functions };
}
function getPermissions_() {
  const sheet = SpreadsheetApp.getActive().getSheetByName("Permissions");
  if (!sheet) throw new Error('Nem található a "Permissions" munkalap.');

  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return { success: true, permissions: [] };

  const header = data[0].map(h => String(h || "").trim());
  const col = (name) => header.indexOf(name);

  const cEmail = col("email");
  const cKey = col("function_key");
  const cAcc = col("access");

  if (cEmail === -1 || cKey === -1 || cAcc === -1) {
    throw new Error('A "Permissions" fejléc hibás. Elvárt oszlopok: email, function_key, access');
  }

  const permissions = [];
  for (let i = 1; i < data.length; i++) {
    const row = data[i];

    const email = String(row[cEmail] || "").trim().toLowerCase();
    const function_key = String(row[cKey] || "").trim();
    const accessRaw = String(row[cAcc] || "").trim().toLowerCase();

    if (!email || !function_key) continue;

    let access = accessRaw;
    if (!["none", "read", "write"].includes(access)) access = "none";

    permissions.push({ email, function_key, access });
  }

  return { success: true, permissions };
}
function getMyPermissions_(email) {
  const sheet = SpreadsheetApp.getActive().getSheetByName("Permissions");
  if (!sheet) return { success: true, permissions: {} };

  const em = String(email || "").trim().toLowerCase();
  if (!em) return { success: true, permissions: {} };

  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return { success: true, permissions: {} };

  const header = (data[0] || []).map(h => String(h || "").trim());
  const col = (name) => header.indexOf(name);

  const cEmail = col("email");
  const cKey = col("function_key");
  const cAcc = col("access");
  if (cEmail === -1 || cKey === -1 || cAcc === -1) {
    throw new Error('A "Permissions" fejléc hibás. Elvárt oszlopok: email, function_key, access');
  }

  const out = {};
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const rowEmail = String(row[cEmail] || "").trim().toLowerCase();
    if (rowEmail !== em) continue;

    const fk = String(row[cKey] || "").trim().toLowerCase();
    if (!fk) continue;

    const accRaw = String(row[cAcc] || "").trim().toLowerCase();
    const acc = (["none", "read", "write"].includes(accRaw) ? accRaw : "none");

    out[fk] = acc;
  }

  return { success: true, permissions: out };
}

function setPermission_(email, function_key, access) {
  const sheet = SpreadsheetApp.getActive().getSheetByName("Permissions");
  if (!sheet) throw new Error('Nem található a "Permissions" munkalap.');

  const em = String(email || "").trim().toLowerCase();
  const fk = String(function_key || "").trim().toLowerCase();
  let acc = String(access || "").trim().toLowerCase();

  if (!em) throw new Error("Hiányzó email.");
  if (!fk) throw new Error("Hiányzó function_key.");
  if (!["none", "read", "write"].includes(acc)) acc = "none";

  const data = sheet.getDataRange().getValues();
  const header = (data[0] || []).map(h => String(h || "").trim());
  const col = (name) => header.indexOf(name);

  const cEmail = col("email");
  const cKey = col("function_key");
  const cAcc = col("access");

  if (cEmail === -1 || cKey === -1 || cAcc === -1) {
    throw new Error('A "Permissions" fejléc hibás. Elvárt oszlopok: email, function_key, access');
  }

  // Keressük a meglévő sort (email + function_key)
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const rowEmail = String(row[cEmail] || "").trim().toLowerCase();
    const rowKey = String(row[cKey] || "").trim().toLowerCase();

    if (rowEmail === em && rowKey === fk) {
      sheet.getRange(i + 1, cAcc + 1).setValue(acc);
      return { success: true, updated: true };
    }
  }

  // Ha nincs meg, új sor
  sheet.appendRow([em, fk, acc]);
  return { success: true, created: true };
}

function requirePermission_(auth, function_key, level) {
  // level: "read" | "write"
  const need = String(level || "").trim().toLowerCase();
  const fk = String(function_key || "").trim();
  if (!fk) throw new Error("Hiányzó function_key.");

  // Admin mindig mehet (megtartjuk a meglévő admin logikát)
  try {
    requireAdmin_(auth);
    return;
  } catch (_) { }

  const email = (auth && auth.email) ? String(auth.email).trim().toLowerCase() : "";
  const acc = getUserAccess_(email, fk);

  const rank = { none: 0, read: 1, write: 2 };
  const ok = (rank[acc] || 0) >= (rank[need] || 0);

  if (!ok) throw new Error("Nincs jogosultság (" + fk + ":" + need + ").");
}

/**
 * GET alapú API (JSONP támogatással).
 * Ezt fogja hívni a file://-ről futó HTML.
 */
function doGet(e) {
  const p = e && e.parameter ? e.parameter : {};
  const action = p.action;
  const callback = p.callback;
  let result;

  try {
    // --- AUTH: login külön, minden más tokenhez kötött ---
    if (action === "login") {
      result = login_(p.email || "", p.password || "");
      return createJsonOrJsonpResponse_(result, callback);
    }

    // loginon kívül mindenhez kell érvényes token
    const auth = requireAuth_(p);

    if (action === "addTransaction") {

      requirePermission_(auth, "tx_create", "write");

      const tx = {
        month: p.month || "",
        date: p.date || "",
        amount: p.amount || "",
        title: p.title || "",
        category: p.category || "",
        payment_type: p.payment_type || "",
        transaction_type: p.transaction_type || "",
        is_shared: p.is_shared || "",
        statement_item: p.statement_item || "",
        paid_by: p.paid_by || ""
      };
      result = addTransaction_(tx);
    } else if (action === "getUsers") {
      requirePermission_(auth, "admin_users", "read");
      result = getUsers_();

    } else if (action === "getPermissions") {
      requirePermission_(auth, "admin_permissions", "read");
      result = getPermissions_();

    } else if (action === "setPermission") {
      requirePermission_(auth, "admin_permissions", "write");
      result = setPermission_(p.email || "", p.function_key || "", p.access || "");

    } else if (action === "getFunctions") {
      requirePermission_(auth, "admin_functions", "read");
      result = getFunctions_();

    } else if (action === "addUser") {
      requirePermission_(auth, "admin_users", "write");
      result = addUser_(p.name || "", p.email || "");


    } else if (action === "getTransactions") {
      requirePermission_(auth, "tx_read", "read");
      result = getTransactions_(p);

    } else if (action === "getValueSets") {
      result = getValueSets_();

    } else if (action === "getValueSetsDetailed") {
      result = getValueSetsDetailed_();

    } else if (action === "updateTransaction") {
      const tx = {
        id: p.id || "",
        month: p.month || "",
        date: p.date || "",
        amount: p.amount || "",
        title: p.title || "",
        category: p.category || "",
        payment_type: p.payment_type || "",
        transaction_type: p.transaction_type || "",
        is_shared: p.is_shared || "",
        statement_item: p.statement_item || "",
        paid_by: p.paid_by || ""
      };
      requirePermission_(auth, "tx_update", "write");
      result = updateTransaction_(tx);
    } else if (action === "bulkMatchTransactions") {
      // jogosultság: ugyanaz, mint update (write)
      requirePermission_(auth, "tx_update", "write");

      // items: JSON string (URL paraméterben)
      const raw = p.items || "[]";
      let items = [];
      try {
        items = JSON.parse(raw);
      } catch (e) {
        result = { success: false, error: "Hibás JSON a 'items' paraméterben." };
        return createJsonOrJsonpResponse_(result, callback);
      }
      // ===== Hardening =====

      if (!Array.isArray(items)) {
        result = { success: false, error: "Az 'items' paraméternek tömbnek kell lennie." };
        return createJsonOrJsonpResponse_(result, callback);
      }

      if (items.length > 500) {
        result = { success: false, error: "Túl sok elem (max 500 engedélyezett egy kérésben)." };
        return createJsonOrJsonpResponse_(result, callback);
      }

      for (var i = 0; i < items.length; i++) {
        var it = items[i];
        if (!it || typeof it !== "object" || !it.id) {
          result = { success: false, error: "Hibás items struktúra az indexen: " + i };
          return createJsonOrJsonpResponse_(result, callback);
        }
      }
      result = bulkMatchTransactions_(items);
    } else if (action === "addValueToSet") {
      result = addValueToSet_(p);

    } else if (action === "getSharedExpenses") {
      requirePermission_(auth, "se_read", "read");
      result = getSharedExpenses_(p);


    } else if (action === "updateSharedExpense") {
      requirePermission_(auth, "se_update", "write");
      result = updateSharedExpense_(p.id, p.field, p.value);

    } else if (action === "updateSharedExpenseRow") {
      requirePermission_(auth, "se_inline_update", "write");
      result = updateSharedExpenseRow_(p);
    }
    else if (action === "deleteTransaction") {
      requirePermission_(auth, "tx_delete", "write");
      result = deleteTransaction_(p.id || "");


    } else if (action === "addSharedExpense") {
      requirePermission_(auth, "se_create", "write");
      result = addSharedExpense_(p);
    } else if (action === "addTransactions") {
      requirePermission_(auth, "tx_import", "write");
      // items: JSON string (URL paraméterben)
      const raw = p.items || "[]";
      let items = [];
      try {
        items = JSON.parse(raw);
      } catch (e) {
        result = { success: false, error: "Hibás JSON a 'items' paraméterben." };
        return createJsonOrJsonpResponse_(result, callback);
      }

      result = addTransactions_(items);
    } else if (action === "addBankTransactions") {
      // jogosultság: egyelőre a meglévő import jogot használjuk
      requirePermission_(auth, "tx_import", "write");

      const raw = p.items || "[]";
      let items = [];
      try {
        items = JSON.parse(raw);
      } catch (e) {
        result = { success: false, error: "Hibás JSON a 'items' paraméterben." };
        return createJsonOrJsonpResponse_(result, callback);
      }

      result = addBankTransactions_(items);
    } else if (action === "getBankTransactions") {
            const cache = CacheService.getScriptCache();
      const cached = cache.get("bank_transactions_list");

      if (cached) {
        try {
          return createJsonOrJsonpResponse_(JSON.parse(cached), callback);
        } catch (e) {}
      }

      // jogosultság: egyelőre ugyanahhoz kötjük, mint az importot (write -> read is ok)

      const sheet = SpreadsheetApp.getActive().getSheetByName("Bank_Transactions");
      if (!sheet) throw new Error('Nem található a "Bank_Transactions" nevű munkalap.');

      const data = sheet.getDataRange().getValues();
      if (!data || data.length < 2) {
        result = { success: true, data: [] };
        return createJsonOrJsonpResponse_(result, callback);
      }

      const header = data[0];
      const tz = Session.getScriptTimeZone();
      // --- BT id -> [TR id, ...] index a Transactions.statement_item alapján ---
      const txSheet = SpreadsheetApp.getActive().getSheetByName("Transactions");
      const btToTxIds = new Map();

      if (txSheet) {
        const txData = txSheet.getDataRange().getValues();
        if (txData && txData.length >= 2) {
          const txHeader = (txData[0] || []).map(h => String(h || "").trim());
          const cTxId = txHeader.indexOf("id");
          const cTxStatement = txHeader.indexOf("statement_item");

          if (cTxId !== -1 && cTxStatement !== -1) {
            for (let r = 1; r < txData.length; r++) {
              const trId = String(txData[r][cTxId] || "").trim();
              const stmtRaw = String(txData[r][cTxStatement] || "").trim();
              if (!trId || !stmtRaw) continue;

              // statement_item lehet "BT-..., BT-..." (vesszővel elválasztva)
              const btIds = stmtRaw.split(",").map(s => s.trim()).filter(Boolean);
              for (const btId of btIds) {
                if (!btToTxIds.has(btId)) btToTxIds.set(btId, []);
                btToTxIds.get(btId).push(trId);
              }
            }
          }
        }
      }
      const rows = data.slice(1).map(row => {
        const obj = {};
        header.forEach((name, idx) => {
          const cell = row[idx];

// dátumok stabil visszaadása
if (cell instanceof Date && !isNaN(cell.getTime())) {
  // created_at esetén idő is kellhet
  const fmt = (name === "created_at")
    ? "yyyy-MM-dd HH:mm:ss"
    : "yyyy-MM-dd";
  obj[name] = Utilities.formatDate(cell, tz, fmt);
} else {
  obj[name] = cell;
}
        });
                // ÚJ: a banki tételhez rendelt tranzakció ID-k (csak kijelzéshez)
        const btId = String(obj.id || "").trim();
        obj.matched_transaction_ids = (btToTxIds.get(btId) || []).join(", ");
        return obj;
      });

      result = { success: true, data: rows };

    } else if (action === "deleteSharedExpense") {
      requirePermission_(auth, "se_delete", "write");
      result = deleteSharedExpense_(p.id || "");

    } else if (action === "whoami") {
      result = { success: true, user: { id: auth.id, name: auth.name, email: auth.email } };
    } else if (action === "getMyPermissions") {
      // Frontend nem küld email paramétert, ezért a session-ből vesszük
      // Ha nálad getMyPermissions_ emailt vár (ahogy most írtad), ez így kompatibilis:
      result = getMyPermissions_(auth.email);

    } else if (action === "logout") {
      result = logout_(p.token || "");

    } else {
      result = { success: false, error: "Ismeretlen action: " + action };
    }



  } catch (err) {
    result = { success: false, error: err.message || String(err) };
  }

  // 🔥 A HELYES HELYEN hívjuk meg a JSONP-t
  return createJsonOrJsonpResponse_(result, callback);
}

function addTransactions_(items) {
  const sheet = SpreadsheetApp.getActive().getSheetByName("Transactions");
  if (!sheet) throw new Error('Nem található a "Transactions" nevű munkalap.');

  if (!Array.isArray(items) || items.length === 0) {
    return { success: true, ok: 0, fail: 0, results: [] };
  }

  // Transactions séma: ugyanaz a sorrend, mint addTransaction_()-ben
  const results = [];
  const rowsToInsert = [];

  for (let i = 0; i < items.length; i++) {
    const p = items[i] || {};

    // minimál validáció (frontenden is van, de itt is kell)
    const date = (p.date || "").toString().trim();
    const amount = (p.amount || "").toString().trim();
    const title = (p.title || "").toString().trim();

    if (!date || !amount || !title) {
      results.push({ index: i, success: false, error: "Hiányzó kötelező mező (date/amount/title)." });
      continue;
    }

    const id = generatePrefixedId_("TR");
    const created_at = new Date();
    const created_by = Session.getActiveUser().getEmail() || "unknown";

    const tx = {
      id,
      month: p.month || "",
      date: p.date || "",
      amount: p.amount || "",
      title: p.title || "",
      category: p.category || "",
      payment_type: p.payment_type || "",
      transaction_type: p.transaction_type || "",
      is_shared: p.is_shared || "",
      statement_item: p.statement_item || "",
      paid_by: p.paid_by || ""
    };

    const row = [
      tx.id,
      tx.month,
      formatDateForStore_(tx.date),
      tx.amount,
      tx.title,
      tx.category,
      tx.payment_type,
      tx.transaction_type,
      tx.is_shared,
      tx.statement_item,
      created_by,
      created_at
    ];

    rowsToInsert.push(row);
    results.push({ index: i, success: true, id: id, _tx: tx }); // _tx: belső, később szinkronhoz
  }

  if (rowsToInsert.length > 0) {
    const startRow = sheet.getLastRow() + 1;
    sheet.getRange(startRow, 1, rowsToInsert.length, rowsToInsert[0].length).setValues(rowsToInsert);
    // TODO: MATCH bank transactions to Transactions by transaction_date + amount, write BT id into Transactions.statement_item
    // --- MATCH PREP: load Transactions sheet + column indexes ---
    const txSheet = SpreadsheetApp.getActive().getSheetByName("Transactions");
    if (!txSheet) throw new Error('Nem található a "Transactions" nevű munkalap.');

    const txData = txSheet.getDataRange().getValues();
    if (!txData || txData.length < 2) {
      // nincs mit párosítani
    } else {
      const txHeader = txData[0].map(h => String(h || "").trim());

      const cTxId = txHeader.indexOf("id");
      const cTxDate = txHeader.indexOf("date");
      const cTxAmount = txHeader.indexOf("amount");
      const cTxStatement = txHeader.indexOf("statement_item");

      if (cTxId === -1 || cTxDate === -1 || cTxAmount === -1 || cTxStatement === -1) {
        throw new Error('A "Transactions" fejléc hibás. Kötelező oszlopok: id, date, amount, statement_item');
      }

      // --- BUILD INDEX: (date|amount) -> tx rows with empty statement_item ---
      const tz = Session.getScriptTimeZone();

      const normDate = (cell) => {
        if (cell instanceof Date && !isNaN(cell.getTime())) {
          return Utilities.formatDate(cell, tz, "yyyy-MM-dd");
        }
        const s = String(cell || "").trim();
        if (!s) return "";
        // ha már yyyy-mm-dd
        const m = s.match(/^(\d{4})-(\d{2})-(\d{2})$/);
        if (m) return s;

        // ha yyyy.mm.dd vagy yyyy.mm.dd.
        const m2 = s.match(/^(\d{4})\.(\d{2})\.(\d{2})\.?$/);
        if (m2) return `${m2[1]}-${m2[2]}-${m2[3]}`;

        // fallback
        const d = new Date(s);
        return isNaN(d.getTime()) ? "" : Utilities.formatDate(d, tz, "yyyy-MM-dd");
      };

      const normAmount = (v) => {
        const s = String(v ?? "").trim().replace(/\s+/g, "").replace(",", ".");
        const n = Number(s);
        if (isNaN(n)) return null;

        // párosítás abs értéken + kerekítve (bank vs tx előjel/kerekítés eltérés ellen)
        const abs = Math.abs(n);
        return Math.round(abs * 100) / 100;
      };


      const keyOf = (dateIso, amountNum) => `${dateIso}|${amountNum}`;

      // map: kulcs -> [txRowIndex1Based, ...] (csak ahol statement_item üres)
      const txMap = new Map();

      for (let r = 1; r < txData.length; r++) {
        const row = txData[r];
        const stmt = String(row[cTxStatement] ?? "").trim();
        if (stmt) continue; // ne írjunk felül meglévőt

        const dIso = normDate(row[cTxDate]);
        const amt = normAmount(row[cTxAmount]);
        if (!dIso || amt == null) continue;

        const k = keyOf(dIso, amt);
        if (!txMap.has(k)) txMap.set(k, []);
        txMap.get(k).push({
          rowIndex1Based: r + 1,
          txId: String(row[cTxId] || "").trim()
        });

      }

      // --- MATCH: loop newly inserted bank rows (rowsToInsert) and write BT id into Transactions.statement_item ---
      const updates = []; // { rowIndex1Based, value }
      let unmatched = 0;
      const unmatchedSamples = [];

      // rowsToInsert: [ id, month, transaction_date, ..., amount, ... ]
      // itt a sémád szerint:
      // 0=id, 2=transaction_date, 12=amount
      for (let i = 0; i < rowsToInsert.length; i++) {
        const bankRow = rowsToInsert[i];
        const btId = String(bankRow[0] || "").trim();
        const btDateIso = normDate(bankRow[2]);
        const btAmt = normAmount(bankRow[12]);

        if (!btId || !btDateIso || btAmt == null) continue;

        const k = keyOf(btDateIso, btAmt);
        const candidates = txMap.get(k);
        if (!candidates || candidates.length === 0) {
          unmatched++;
          if (unmatchedSamples.length < 10) {
            unmatchedSamples.push({ btId, btDateIso, btAmt });
          }
          continue;
        }

        // determinisztikus: a legkisebb ID-jű tranzakciót választjuk
        candidates.sort((a, b) => a.txId.localeCompare(b.txId, "hu"));

        const chosen = candidates.shift();
        if (!chosen) continue;

        updates.push({ rowIndex1Based: chosen.rowIndex1Based, value: btId });

      }

      // --- APPLY updates ---
      for (const u of updates) {
        txSheet.getRange(u.rowIndex1Based, cTxStatement + 1).setValue(u.value);
      }
      const matched = updates.length;
    }

    // Value_Sets frissítés + shared sync (ugyanaz, mint addTransaction_)
    for (const r of results) {
      if (!r.success) continue;
      const tx = r._tx;

      addValueIfMissing_("title_values", tx.title);
      addValueIfMissing_("category_values", tx.category);
      addValueIfMissing_("payment_type_values", tx.payment_type);
      addValueIfMissing_("transaction_type_values", tx.transaction_type);

      syncSharedExpenseForTransaction_({
        id: tx.id,
        month: tx.month,
        date: tx.date,
        amount: tx.amount,
        title: tx.title,
        is_shared: tx.is_shared,
        paid_by: tx.paid_by
      });

      delete r._tx; // ne küldjük vissza
    }
  }

  const ok = results.filter(x => x.success).length;
  const fail = results.length - ok;

  return { success: true, ok, fail, matched: (typeof matched === "number" ? matched : 0), results };
}
function addBankTransactions_(items) {
  const sheet = SpreadsheetApp.getActive().getSheetByName("Bank_Transactions");
  if (!sheet) throw new Error('Nem található a "Bank_Transactions" nevű munkalap.');

  if (!Array.isArray(items) || items.length === 0) {
    return { success: true, ok: 0, fail: 0, results: [] };
  }

  const results = [];
  const rowsToInsert = [];

  for (let i = 0; i < items.length; i++) {
    const p = items[i] || {};

    // minimál validáció: a banki sor értelmezhető legyen
    const txDate = (p.transaction_date || "").toString().trim();
    const amount = (p.amount || "").toString().trim();

    if (!txDate || !amount) {
      results.push({ index: i, success: false, error: "Hiányzó kötelező mező (transaction_date/amount)." });
      continue;
    }

    const id = generatePrefixedId_("BT");
    const created_at = new Date();
    const created_by = Session.getActiveUser().getEmail() || "unknown";

    // month: ha nem jön a front-endről, képezzük transaction_date-ből
    const month = (p.month || "").toString().trim() || deriveMonthFromDate_(txDate);

    // A Bank_Transactions fejléc szerinti sorrend (amit most hoztál létre)
    const row = [
      id,                                 // id
      month,                              // month
      formatDateForStore_(txDate),        // transaction_date
      formatDateForStore_(p.posting_date || ""), // posting_date
      p.type || "",                       // type
      p.direction || "",                  // direction
      p.partner_name || "",               // partner_name
      p.partner_account || "",            // partner_account
      p.spend_category || "",             // spend_category
      p.memo || "",                       // memo
      p.account_name || "",               // account_name
      p.account_number || "",             // account_number
      p.amount || "",                     // amount
      p.currency || "",                   // currency
      p.source_file || "",                // source_file
      p.import_batch_id || "",            // import_batch_id
      created_by,                         // created_by
      created_at                          // created_at
    ];

    rowsToInsert.push(row);
    results.push({ index: i, success: true, id });
  }
  // --- INSERT new bank rows into Bank_Transactions sheet ---
  if (rowsToInsert.length > 0) {
    const startRow = sheet.getLastRow() + 1;
    sheet.getRange(startRow, 1, rowsToInsert.length, rowsToInsert[0].length).setValues(rowsToInsert);
  }

  // --- APPLY updates (batch, contiguous chunks) ---
  updates.sort((a, b) => a.rowIndex1Based - b.rowIndex1Based);

  let i = 0;
  while (i < updates.length) {
    const start = updates[i].rowIndex1Based;
    let end = start;

    const values = [[updates[i].value]];
    i++;

    // egymást követő sorok egy chunkba
    while (i < updates.length && updates[i].rowIndex1Based === end + 1) {
      end = updates[i].rowIndex1Based;
      values.push([updates[i].value]);
      i++;
    }

    // statement_item egyetlen oszlop → width = 1
    txSheet.getRange(start, cTxStatement + 1, (end - start + 1), 1).setValues(values);
  }


  const ok = results.filter(x => x.success).length;
  const fail = results.length - ok;

  return {
    success: true,
    ok,
    fail,
    matched: (typeof matched === "number" ? matched : (Array.isArray(updates) ? updates.length : 0)),
    unmatched,
    unmatched_samples: unmatchedSamples,
    results
  };
}

/**
 * Tömeges összerendelés: csak a statement_item oszlop frissítése ID alapján.
 * items: [{ id: "TX-...", statement_item: "BT-..." }, ...]
 */
function bulkMatchTransactions_(items) {
  if (!Array.isArray(items) || items.length === 0) {
    return { success: true, ok: 0, fail: 0, results: [] };
  }

  const sheet = SpreadsheetApp.getActive().getSheetByName("Transactions");
  if (!sheet) throw new Error('Nem található a "Transactions" nevű munkalap.');

  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return { success: true, ok: 0, fail: items.length, results: [] };

  const header = (data[0] || []).map(h => String(h || "").trim());
  // Preferált: fejléc alapján, fallback: a jelenlegi struktúrád szerint statement_item a 10. oszlop (index 9)
  let cStatement = header.indexOf("statement_item");
  if (cStatement === -1) cStatement = 9;

  // ID oszlop: a jelenlegi kódod szerint első oszlop (index 0)
  const cId = 0;

  // ID -> rowIndex (1-indexelt sheet sor)
  const rowById = new Map();
  for (let i = 1; i < data.length; i++) {
    const id = String(data[i][cId] || "").trim();
    if (id) rowById.set(id, i + 1);
  }

  const results = [];
  let ok = 0, fail = 0;

  // Csoportosítjuk a beállítandó értékeket sorindex szerint
  const updates = [];
  for (const it of items) {
    const id = String(it?.id ?? "").trim();
    const bankId = String(it?.statement_item ?? "").trim();
    if (!id) {
      fail++;
      results.push({ success: false, id, error: "Hiányzó tranzakció ID." });
      continue;
    }

    const rowIndex = rowById.get(id);
    if (!rowIndex) {
      fail++;
      results.push({ success: false, id, error: "Nincs ilyen tranzakció ID." });
      continue;
    }

    updates.push({ rowIndex, id, bankId });
  }

  // Sorindex szerint rendezve (stabilabb, kevesebb szétugrás)
  updates.sort((a, b) => a.rowIndex - b.rowIndex);

  // Egyesével írjuk (biztos, egyszerű, és nem nyúl más oszlophoz).
  // (Ha akarod, következő körben optimalizáljuk tartományokra.)
  for (const u of updates) {
    try {
      sheet.getRange(u.rowIndex, cStatement + 1).setValue(u.bankId);
      ok++;
      results.push({ success: true, id: u.id, statement_item: u.bankId });
    } catch (e) {
      fail++;
      results.push({ success: false, id: u.id, error: String(e?.message || e) });
    }
  }

  return { success: true, ok, fail, results };
}
function deleteTransaction_(id) {
  const txId = String(id || "").trim();
  if (!txId) return { success: false, error: "Hiányzó id" };

  // 1) Transactions sor törlése ID alapján
  const txSheet = SpreadsheetApp.getActive().getSheetByName("Transactions");
  if (!txSheet) throw new Error('Nem található a "Transactions" nevű munkalap.');

  const txData = txSheet.getDataRange().getValues();
  const txHeader = txData[0];
  const txIdCol = 0; // ID az első oszlopban

  let txRowIndex = -1;
  for (let i = 1; i < txData.length; i++) {
    if (String(txData[i][txIdCol]).trim() === txId) {
      txRowIndex = i + 1; // sheet 1-indexelt
      break;
    }
  }
  if (txRowIndex === -1) {
    return { success: false, error: "Nincs ilyen tranzakció ID: " + txId };
  }

  txSheet.deleteRow(txRowIndex);

function addValueIfMissing_(columnName, value) {
  if (!value) return;

  const sheet = SpreadsheetApp.getActive().getSheetByName("Value_Sets");
  const data = sheet.getDataRange().getValues();
  const header = data[0];

  const colIndex = header.indexOf(columnName);
  if (colIndex === -1) return;

  // Már létezik?
  for (let i = 1; i < data.length; i++) {
    if (
      String(data[i][colIndex]).trim().toLowerCase() ===
      String(value).trim().toLowerCase()
    ) {
      return;
    }
  }

  // Következő üres sor keresése ebben az oszlopban
  let insertRow = 1;
  for (let i = 1; i < data.length; i++) {
    if (data[i][colIndex] === "" || data[i][colIndex] === null) {
      insertRow = i + 1; // Google Sheets 1-indexelt
      break;
    }
    insertRow = i + 2; // ha nincs több üres cella
  }

  // Beszúrjuk az értéket az adott oszlop megfelelő cellájába
  sheet.getRange(insertRow, colIndex + 1).setValue(value);
}
/**
 * Új érték hozzáadása a Value_Sets lap megfelelő oszlopához.
 */
function addValueToSet_(p) {
  const sheet = SpreadsheetApp.getActive().getSheetByName("Value_Sets");
  const setName = p.set;
  const value = (p.value || "").trim();

  if (!setName || !value) {
    return { success: false, message: "Hiányzó set vagy érték." };
  }

  const header = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const col = header.indexOf(setName);

  if (col === -1) {
    return { success: false, message: "Nincs ilyen oszlop: " + setName };
  }

  const lastRow = sheet.getLastRow();
  let existingValues = [];
  let insertRow = 2; // alapértelmezés: a fejléc alatti első sor

  if (lastRow > 1) {
    // Teljes oszlopértékek beolvasása (fejléc alatti részen)
    const colValues = sheet
      .getRange(2, col + 1, lastRow - 1, 1)
      .getValues();

    // Meglévő értékek normalizálása (duplikáció ellenőrzéshez)
    existingValues = colValues
      .map(r => (r[0] || "").toString().trim().toLowerCase())
      .filter(v => v !== "");

    // Következő üres sor keresése ebben az oszlopban
    insertRow = 2;
    for (let i = 0; i < colValues.length; i++) {
      const cell = colValues[i][0];
      if (!cell || String(cell).trim() === "") {
        insertRow = 2 + i; // első üres sor
        break;
      }
      insertRow = 2 + i + 1; // ha eddig minden nem üres, akkor a következő sor
    }
  }

  // Már létezik?
  if (existingValues.includes(value.toLowerCase())) {
    return { success: true, message: "Érték már létezik." };
  }

  // Új érték beszúrása az oszlop következő üres sorába
  sheet.getRange(insertRow, col + 1).setValue(value);

  return { success: true, message: "Érték hozzáadva." };
}

/**
 * Megosztott tranzakció szinkronizálása a Shared_Expenses lapra.
 */
function syncSharedExpenseForTransaction_(tx) {
  const sheet = SpreadsheetApp.getActive().getSheetByName("Shared_Expenses");
  if (!sheet) return;

  const data = sheet.getDataRange().getValues();
  const header = data[0];

  // Oszlopindexek (új, jelenlegi séma szerint)
  const colId = header.indexOf("id");
  const colCreatedAt = header.indexOf("created_at");
  const colTxId = header.indexOf("transaction_id");
  const colMonth = header.indexOf("month");
  const colDate = header.indexOf("date");
  const colTitle = header.indexOf("title");
  const colAmount = header.indexOf("amount");
  const colPaidBy = header.indexOf("paid_by");

  const colZsoltiAmount = header.indexOf("Zsolti_amount");
  const colDoriAmount = header.indexOf("Dori_amount");
  const colRemaining = header.indexOf("remaining_amount");
  const colZsoltiBal = header.indexOf("Zsolti_balance");
  const colDoriBal = header.indexOf("Dori_balance");
  const colBalanceImpact = header.indexOf("balance_impact");

  const colNotes = header.indexOf("notes");

  // Meglévő sor keresése transaction_id alapján
  let rowIndex = -1;
  let existingRow = null;

  for (let i = 1; i < data.length; i++) {
    if (String(data[i][colTxId]).trim() === String(tx.id).trim()) {
      rowIndex = i + 1;      // 1-indexelt sheet sor
      existingRow = data[i]; // 0-indexelt tömb
      break;
    }
  }

  // Ha nem megosztott: töröljük a hozzá tartozó Shared_Expenses sort (ha van)
  const isShared = String(tx.is_shared || "").trim().toLowerCase() === "x";
  if (!isShared) {
    if (rowIndex !== -1) sheet.deleteRow(rowIndex);
    return;
  }

  // Shared_Expenses lapon az amount mindig pozitív legyen
  const amountAbs = Math.abs(Number(tx.amount) || 0);

  // paid_by: új sornál tx.paid_by (ha nincs: Dóri), frissítésnél nem írjuk felül
  const fallbackPaidBy = (tx.paid_by && String(tx.paid_by).trim()) ? tx.paid_by : "Dóri";
  const paidByForCalc =
    (rowIndex === -1)
      ? fallbackPaidBy
      : ((colPaidBy !== -1 && existingRow) ? (existingRow[colPaidBy] || fallbackPaidBy) : fallbackPaidBy);

  // Importált shared tételnél alapból nincs egyéni megosztás
  const zAbs = 0;
  const dAbs = 0;

  // Számítás (alap shared logika)
  const remaining = amountAbs - dAbs - zAbs; // = amountAbs
  const halfRemaining = remaining / 2;

  const zsoltiBal = halfRemaining + zAbs; // = amountAbs/2
  const doriBal = halfRemaining + dAbs; // = amountAbs/2

  // balance_impact: a NEM fizető fél egyenlege (előjel nélkül)
  let balanceImpact = 0;
  const paidByNorm = String(paidByForCalc || "").trim().toLowerCase();
  if (paidByNorm === "dóri" || paidByNorm === "dori") {
    balanceImpact = zsoltiBal; // Dóri fizetett → Zsolti tartozása
  } else if (paidByNorm === "zsolti") {
    balanceImpact = doriBal;   // Zsolti fizetett → Dóri tartozása
  }

  if (rowIndex === -1) {
    // Új sor beszúrása
    const newRow = new Array(header.length).fill("");

    if (colId !== -1) newRow[colId] = generatePrefixedId_("SE");
    if (colCreatedAt !== -1) newRow[colCreatedAt] = new Date();
    if (colTxId !== -1) newRow[colTxId] = tx.id;

    if (colMonth !== -1) newRow[colMonth] = tx.month;
    if (colDate !== -1) newRow[colDate] = formatDateForStore_(tx.date);
    if (colTitle !== -1) newRow[colTitle] = tx.title;
    if (colAmount !== -1) newRow[colAmount] = amountAbs;
    if (colPaidBy !== -1) newRow[colPaidBy] = fallbackPaidBy;

    if (colZsoltiAmount !== -1) newRow[colZsoltiAmount] = 0;
    if (colDoriAmount !== -1) newRow[colDoriAmount] = 0;

    if (colRemaining !== -1) newRow[colRemaining] = remaining;
    if (colZsoltiBal !== -1) newRow[colZsoltiBal] = zsoltiBal;
    if (colDoriBal !== -1) newRow[colDoriBal] = doriBal;
    if (colBalanceImpact !== -1) newRow[colBalanceImpact] = balanceImpact;

    // notes-t nem töltünk automatikusan, maradhat üres
    sheet.appendRow(newRow);

  } else {
    // Meglévő sor frissítése (paid_by-t nem írjuk felül)
    const row = existingRow.slice();

    if (colMonth !== -1) row[colMonth] = tx.month;
    if (colDate !== -1) row[colDate] = formatDateForStore_(tx.date);
    if (colTitle !== -1) row[colTitle] = tx.title;
    if (colAmount !== -1) row[colAmount] = amountAbs;

    // Alap megosztási mezők importált tételnél 0-ra
    if (colZsoltiAmount !== -1) row[colZsoltiAmount] = 0;
    if (colDoriAmount !== -1) row[colDoriAmount] = 0;

    // Újraszámolt mezők
    if (colRemaining !== -1) row[colRemaining] = remaining;
    if (colZsoltiBal !== -1) row[colZsoltiBal] = zsoltiBal;
    if (colDoriBal !== -1) row[colDoriBal] = doriBal;
    if (colBalanceImpact !== -1) row[colBalanceImpact] = balanceImpact;

    sheet.getRange(rowIndex, 1, 1, row.length).setValues([row]);
  }
}

/**
 * Shared_Expenses egy mezőjének manuális módosítása + automatikus újraszámítás.
 * Elvárt oszlopok (a sheet fejlécben): 
 * id, amount, paid_by, Zsolti_amount, Dori_amount, remaining_amount,
 * Zsolti_balance, Dori_balance, balance_impact, notes
 */
function updateSharedExpense_(id, field, value) {
  const sheet = SpreadsheetApp.getActive().getSheetByName("Shared_Expenses");
  if (!sheet) {
    return { success: false, error: 'Nem található a "Shared_Expenses" munkalap.' };
  }

  const data = sheet.getDataRange().getValues();
  const header = data[0];

  // Kötelező oszlopok indexei
  const colId = header.indexOf("id");
  const colAmount = header.indexOf("amount");
  const colTitle = header.indexOf("title");
  const colPaidBy = header.indexOf("paid_by");
  const colZsoltiAmount = header.indexOf("Zsolti_amount");
  const colDoriAmount = header.indexOf("Dori_amount");
  const colRemaining = header.indexOf("remaining_amount");
  const colZsoltiBalance = header.indexOf("Zsolti_balance");
  const colDoriBalance = header.indexOf("Dori_balance");
  const colBalanceImpact = header.indexOf("balance_impact");
  const colNotes = header.indexOf("notes");

  // Csak ezek a mezők legyenek UI-ból írhatók
  let targetCol = null;
  if (field === "paid_by") targetCol = colPaidBy;
  if (field === "Zsolti_amount") targetCol = colZsoltiAmount;
  if (field === "Dori_amount") targetCol = colDoriAmount;
  if (field === "notes") targetCol = colNotes;

  if (targetCol === null || targetCol === -1) {
    return { success: false, error: "Nem módosítható mező vagy hiányzó oszlop: " + field };
  }

  // Sor megkeresése ID alapján
  let rowIndex = -1;
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][colId]).trim() === String(id).trim()) {
      rowIndex = i + 1; // sheet 1-indexelt
      break;
    }
  }
  if (rowIndex === -1) {
    return { success: false, error: "Nincs ilyen ID: " + id };
  }

  // 1) Módosított mező mentése (amount mezők mindig abs értékkel tárolódjanak)
  let storeValue = value;

  if (field === "Zsolti_amount" || field === "Dori_amount") {
    storeValue = Math.abs(Number(value) || 0);
  }

  sheet.getRange(rowIndex, targetCol + 1).setValue(storeValue);


  // 2) Újraszámítás ugyanarra a sorra (a frissen írt értékkel együtt)
  const rowValues = sheet.getRange(rowIndex, 1, 1, header.length).getValues()[0];
  const titleVal = colTitle !== -1 ? String(rowValues[colTitle] || "") : "";
  const isSettlement = titleVal.trim().toLowerCase().includes("törleszt");



  const amount = Math.abs(Number(colAmount !== -1 ? rowValues[colAmount] : 0) || 0);

  const paidBy = (colPaidBy !== -1 ? String(rowValues[colPaidBy]) : "")
    .trim()
    .toLowerCase();

  // Ha épp ezt a mezőt írtuk, akkor a paraméter value a mérvadó
  const zsoltiAmount =
    colZsoltiAmount === -1
      ? 0
      : Math.abs(Number(field === "Zsolti_amount" ? storeValue : rowValues[colZsoltiAmount]) || 0);

  const doriAmount =
    colDoriAmount === -1
      ? 0
      : Math.abs(Number(field === "Dori_amount" ? storeValue : rowValues[colDoriAmount]) || 0);

  let remaining = "";
  let zsoltiBal = "";
  let doriBal = "";
  let balanceImpact = 0;

  if (isSettlement) {
    balanceImpact = amount;
  } else {
    remaining = amount - doriAmount - zsoltiAmount;
    const halfRemaining = remaining / 2;

    zsoltiBal = halfRemaining + zsoltiAmount;
    doriBal = halfRemaining + doriAmount;

    let bi = 0;
    if (paidBy === "dóri" || paidBy === "dori") bi = zsoltiBal;
    else if (paidBy === "zsolti") bi = doriBal;
    balanceImpact = bi;
  }



  // Visszaírások (csak ha az oszlop létezik)
  if (colRemaining !== -1) sheet.getRange(rowIndex, colRemaining + 1).setValue(remaining);
  if (colZsoltiBalance !== -1) sheet.getRange(rowIndex, colZsoltiBalance + 1).setValue(zsoltiBal);
  if (colDoriBalance !== -1) sheet.getRange(rowIndex, colDoriBalance + 1).setValue(doriBal);
  if (colBalanceImpact !== -1) sheet.getRange(rowIndex, colBalanceImpact + 1).setValue(balanceImpact);


  return { success: true, id: id, field: field, value: value };
}

  const colPaidBy = header.indexOf("paid_by");
  const colZsoltiAmount = header.indexOf("Zsolti_amount");
  const colDoriAmount = header.indexOf("Dori_amount");
  const colRemaining = header.indexOf("remaining_amount");
  const colZsoltiBalance = header.indexOf("Zsolti_balance");
  const colDoriBalance = header.indexOf("Dori_balance");
  const colBalanceImpact = header.indexOf("balance_impact");
  const colNotes = header.indexOf("notes");

  const id = String(p.id || "").trim();
  if (!id) return { success: false, error: "Hiányzó id (szerkesztéshez kötelező)." };
  if (colId === -1) return { success: false, error: 'Hiányzó "id" oszlop a Shared_Expenses lapon.' };

  // sor keresése ID alapján
  const data = sheet.getDataRange().getValues();
  let rowIndex = -1;
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][colId]).trim() === id) {
      rowIndex = i + 1; // sheet 1-indexelt
      break;
    }
  }
  if (rowIndex === -1) return { success: false, error: "Nincs ilyen ID: " + id };

  // bejövő értékek
  const dateRaw = String(p.date || "").trim();
  const title = String(p.title || "").trim();
  const notes = String(p.notes || "").trim();
  const paidByRaw = String(p.paid_by || "").trim();
  const paidByNorm = paidByRaw.toLowerCase();

  const amountAbs = Math.abs(Number(p.amount) || 0);

  // törlesztés felismerés: settlement flag vagy cím alapján
  const isSettlement = String(p.settlement || "").trim().toLowerCase() === "x"
    || title.toLowerCase().includes("törleszt");

  const zAbs = isSettlement ? 0 : Math.abs(Number(p.Zsolti_amount) || 0);
  const dAbs = isSettlement ? 0 : Math.abs(Number(p.Dori_amount) || 0);

  // számítások
  let remaining = "";
  let zsoltiBal = "";
  let doriBal = "";
  let balanceImpact = 0;

  if (isSettlement) {
    balanceImpact = amountAbs; // törlesztés: teljes összeg
  } else {
    remaining = amountAbs - dAbs - zAbs;
    const halfRemaining = remaining / 2;

    zsoltiBal = halfRemaining + zAbs;
    doriBal = halfRemaining + dAbs;

    if (paidByNorm === "dóri" || paidByNorm === "dori") {
      balanceImpact = zsoltiBal;
    } else if (paidByNorm === "zsolti") {
      balanceImpact = doriBal;
    } else {
      balanceImpact = 0;
    }
  }

  // visszaírás (csak létező oszlopokba)
  if (colMonth !== -1) sheet.getRange(rowIndex, colMonth + 1).setValue(p.month || deriveMonthFromDate_(dateRaw));
  if (colDate !== -1) sheet.getRange(rowIndex, colDate + 1).setValue(formatDateForStore_(dateRaw));
  if (colTitle !== -1) sheet.getRange(rowIndex, colTitle + 1).setValue(title);
  if (colAmount !== -1) sheet.getRange(rowIndex, colAmount + 1).setValue(amountAbs);
  if (colPaidBy !== -1) sheet.getRange(rowIndex, colPaidBy + 1).setValue(paidByRaw);
  if (colZsoltiAmount !== -1) sheet.getRange(rowIndex, colZsoltiAmount + 1).setValue(zAbs);
  if (colDoriAmount !== -1) sheet.getRange(rowIndex, colDoriAmount + 1).setValue(dAbs);
  if (colNotes !== -1) sheet.getRange(rowIndex, colNotes + 1).setValue(notes);

  if (colRemaining !== -1) sheet.getRange(rowIndex, colRemaining + 1).setValue(remaining);
  if (colZsoltiBalance !== -1) sheet.getRange(rowIndex, colZsoltiBalance + 1).setValue(zsoltiBal);
  if (colDoriBalance !== -1) sheet.getRange(rowIndex, colDoriBalance + 1).setValue(doriBal);
  if (colBalanceImpact !== -1) sheet.getRange(rowIndex, colBalanceImpact + 1).setValue(balanceImpact);

  return { success: true, id: id };
}

function deleteSharedExpense_(id) {
  const seId = String(id || "").trim();
  if (!seId) return { success: false, error: "Hiányzó id" };

  const seSheet = SpreadsheetApp.getActive().getSheetByName("Shared_Expenses");
  if (!seSheet) throw new Error('Nem található a "Shared_Expenses" nevű munkalap.');

  const data = seSheet.getDataRange().getValues();
  if (data.length < 2) return { success: false, error: "Nincs törölhető adat." };

  // Feltételezés: az "id" az első oszlopban van (a projektedben eddig így használtuk).
  const idCol = 0;

  let rowIndex = -1;
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][idCol]).trim() === seId) {
      rowIndex = i + 1; // sheet 1-indexelt
      break;
    }
  }

  if (rowIndex === -1) {
    return { success: false, error: "Nincs ilyen megosztott tétel ID: " + seId };
  }

  // --- ÚJ: csak törlesztés vagy paid_by = Zsolti törölhető ---
  const header = data[0];
  const titleCol = header.indexOf("title");
  const paidByCol = header.indexOf("paid_by");

  const row = data[rowIndex - 1]; // data 0-indexelt, sheet 1-indexelt
  const title = (titleCol !== -1 ? String(row[titleCol] || "") : "").trim().toLowerCase();
  const paidBy = (paidByCol !== -1 ? String(row[paidByCol] || "") : "").trim().toLowerCase();

  const isSettlement = title.includes("törleszt");
  const isPaidByZsolti = (paidBy === "zsolti");

  if (!isSettlement && !isPaidByZsolti) {
    return { success: false, error: "Csak törlesztés vagy paid_by = Zsolti tétel törölhető." };
  }

  seSheet.deleteRow(rowIndex);
  return { success: true, id: seId };

}
/**
 * Dátum egységes tárolási formátumra alakítása: YYYY.MM.DD.
 * Időzóna-biztos: a "YYYY-MM-DD" inputot NEM Date-objektummal parse-oljuk.
 */
function formatDateForStore_(value) {
  if (!value) return "";

  const s = String(value).trim();

  // 1) Ha a frontend <input type="date"> formátumot küldi: "YYYY-MM-DD"
  //    Ne Date()-tel parse-oljuk (időzóna miatt csúszhat), hanem stringből alakítsuk.
  const m = s.match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if (m) return `${m[1]}.${m[2]}.${m[3]}.`;

  // 2) Egyéb eset: Date/ISO string -> Utilities.formatDate a script időzónájával
  const d = (Object.prototype.toString.call(value) === "[object Date]") ? value : new Date(value);
  if (isNaN(d.getTime())) return s;

  return Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy.MM.dd.");
}

function deriveMonthFromDate_(value) {
  if (!value) return "";
  const d = new Date(value);
  if (isNaN(d.getTime())) return "";
  const y = d.getFullYear();
  const m = String(d.getMonth() + 1).padStart(2, "0");
  return y + "" + m; // YYYYMM
}
